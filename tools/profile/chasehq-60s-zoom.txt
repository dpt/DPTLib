# chasehq-60s-zoom.txt -- chasehq-60s.txt at Double size
#
# The same timed 60 s run, with the window switched to Double size first.
# Build and run it as chasehq-60s.txt describes, naming this script
# instead. The redraw time it logs is about twice the normal size's, since
# each redraw copies four times the pixels.

wait 50
click select "Chase H.Q." 10 10

# Double size, from the window's menu. Its item sits at screen 135,180
# while the window has its default position
wait 5
click menu "Chase H.Q." 100 100
wait 5
click select 135 180
wait 15

# options: 3 = Kempston. The menu waits for every key to be released
sleep 8
key return
sleep 0.5
keyup return
sleep 1
key 3
sleep 0.5
keyup 3

# fire starts a game, then fire again skips the 15 s or so of briefing;
# up (Kempston up) then holds accelerate
sleep 8
key .
sleep 0.5
keyup .
sleep 2
key .
sleep 0.5
keyup .
sleep 1
key up

sleep 60
quit
